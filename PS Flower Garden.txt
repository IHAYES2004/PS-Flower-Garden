<script>
const reasons = [
"because i was her hero once.",
"because she is absolutely beautiful.",
"the way her smile completely brightens up my darkest days.",
"how passionate and dedicated she is about the things and people she loves.",
"because she has the kindest, most loveable lady Yaya has ever known.",
"the sound of her laugh and grin honestly my favorite sound in the world.",
"how safe and completely at home I feel whenever she is around.",
"she remembers the little details.",
"her incredible strength and the grace she shows or hide.",
"the cute little expressions she makes when she's super focused.",
"because she brings out the absolute best version of who I am, wish to do the same.",
"the gentle, comforting way she offers support when things get tough.",
"how effortlessly clever, and endlessly fun she is.",
"the beautiful, unique perspective she brings to everyones life.",
"simply because she is unconditionally herself, and she is perfect to me."
];

const forest = document.getElementById('forest');
const activeNote = document.getElementById('activeNote');
const flowerElementsArray = [];
let nextFlowerIndex = 0;

// Dynamically generate the 15 flower components
reasons.forEach((text, i) => {
const flowerItem = document.createElement('div');
flowerItem.className = 'flower-item';

const leftPosition = 5 + (i * 6.4);
const stemHeight = 120 + (i % 3 * 50) + (Math.random() * 20);

flowerItem.style.left = ${leftPosition}%; flowerItem.style.bottom = ``${10 + (i % 2 * 15)}px;
flowerItem.style.setProperty('--stem-height', ${stemHeight}px`);
flowerItem.style.zIndex = Math.floor(stemHeight);

flowerItem.innerHTML = &lt;!-- Firefly Light tracking --&gt; &lt;div class="flower-Light" style="top: -60px; left:${Math.random() * 20 - 10}px;"></div>
<div class="flower-head">
<div class="flower-petal_8"></div>
<div class="flower-petal_8"></div>
<div class="flower-petal_8"></div>
<div class="flower-petal_8"></div>
<div class="flower-petal_8"></div>
<div class="flower-petal_8"></div>
<div class="flower-petal_8"></div>
<div class="flower-petal_8"></div>
<div class="flower-center"></div>
</div>
<div class="flower-stem"></div>
`;

// Update message panel display content
const updateMessage = () => {
activeNote.innerHTML = <strong&gt;Reason #${i + 1}:</strong> I like Sonja ${text};
if(i === 0 || i === 1) {
activeNote.style.borderColor = "#ffe4a0";
} else {
activeNote.style.borderColor = "var(--neon-pink)";
}
};

flowerItem.addEventListener('mouseenter', () => {
if(flowerItem.classList.contains('revealed')) updateMessage();
});
flowerItem.addEventListener('click', (e) => {
e.stopPropagation(); // Stop background trigger override logic
if(flowerItem.classList.contains('revealed')) updateMessage();
});

forest.appendChild(flowerItem);
flowerElementsArray.push(flowerItem);
});

// Click management trigger logic tracking sequence
const growNextFlower = () => {
if (nextFlowerIndex < flowerElementsArray.length) {
const flowerToReveal = flowerElementsArray[nextFlowerIndex];
flowerToReveal.classList.add('revealed');

// Auto trigger text display update for the newly risen flower
const text = reasons[nextFlowerIndex];
activeNote.innerHTML = <strong&gt;Reason #${nextFlowerIndex + 1}:</strong> I like Sonja ${text};
if(nextFlowerIndex === 0 || nextFlowerIndex === 1) {
activeNote.style.borderColor = "#ffe4a0";
} else {
activeNote.style.borderColor = "var(--neon-pink)";
}

nextFlowerIndex++;
} else {
activeNote.innerHTML = "✨ The entire glowing forest has completely bloomed for Sonja! ✨";
activeNote.style.borderColor = "#ffe4a0";
}
};

// Listen to screen/box clicks anywhere to build the sequential step progress chain
document.body.addEventListener('click', growNextFlower);
activeNote.addEventListener('click', (e) => e.stopPropagation()); // prevent block click side effects
</script>
